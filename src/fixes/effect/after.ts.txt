export const leaseRules: LeaseRules = {
  renew: (table, owner, token, now) =>
    live(table, now)?.token === token
      ? [true, { ...table, lease: { owner, token, expiresAt: now + ttl } }]
      : [false, table],
  release: (table, token) => (table.lease?.token === token ? { ...table, lease: null } : table),
};
