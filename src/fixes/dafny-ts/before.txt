export function indexOfId(ids: readonly number[], target: number): number {
  let lo = 0;
  let hi = ids.length - 1;
  while (lo < hi) {
    const mid = Math.floor((lo + hi) / 2);
    if (ids[mid] < target) lo = mid + 1;
    else if (ids[mid] > target) hi = mid - 1;
    else return mid;
  }
  return -1;
}
