---- MODULE PaymentHold ----
\* A card hold on one order. A manager may approve it (capture the money),
\* and a 24h timer releases it back to the customer. Both handlers share one flag.

VARIABLES settled, captured, released, pc

vars == <<settled, captured, released, pc>>

Init ==
    /\ settled = FALSE
    /\ captured = FALSE
    /\ released = FALSE
    /\ pc = [h \in {"approve", "expire"} |-> "idle"]

\* approve() starts: if the hold is open, sets the flag, then awaits capture().
StartApprove ==
    /\ pc["approve"] = "idle"
    /\ pc' = [pc EXCEPT !["approve"] = IF settled THEN "done" ELSE "capturing"]
    /\ settled' = TRUE
    /\ UNCHANGED <<captured, released>>

\* capture() returns.
CaptureOk ==
    /\ pc["approve"] = "capturing"
    /\ captured' = TRUE
    /\ pc' = [pc EXCEPT !["approve"] = "done"]
    /\ UNCHANGED <<settled, released>>

\* capture() throws (card declined, bank timeout). approve() rethrows.
CaptureFails ==
    /\ pc["approve"] = "capturing"
    /\ pc' = [pc EXCEPT !["approve"] = "done"]
    /\ UNCHANGED <<settled, captured, released>>

\* The 24h timer fires: if the hold is open, expire() sets the flag, then awaits release().
StartExpire ==
    /\ pc["expire"] = "idle"
    /\ pc' = [pc EXCEPT !["expire"] = IF settled THEN "done" ELSE "releasing"]
    /\ settled' = TRUE
    /\ UNCHANGED <<captured, released>>

\* release() returns.
ReleaseOk ==
    /\ pc["expire"] = "releasing"
    /\ released' = TRUE
    /\ pc' = [pc EXCEPT !["expire"] = "done"]
    /\ UNCHANGED <<settled, captured>>

Next == StartApprove \/ CaptureOk \/ CaptureFails \/ StartExpire \/ ReleaseOk

\* The timer always fires and awaited calls always return; the manager may never approve.
Spec ==
    /\ Init /\ [][Next]_vars
    /\ WF_vars(StartExpire) /\ WF_vars(ReleaseOk) /\ WF_vars(CaptureOk \/ CaptureFails)

\* Never take the money and also give the hold back.
NotBoth == ~(captured /\ released)

\* Every hold ends: the money is either taken or given back.
Settles == <>(captured \/ released)
====
